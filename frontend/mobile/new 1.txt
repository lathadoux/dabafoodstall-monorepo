# Définissez ici le chemin exact de votre dossier d'installation Java JDK
$javaPath = "C:\Program Files\Eclipse Adoptium\jdk-21.0.12.101-hotspot" # Exemple de chemin

# Enregistrement de JAVA_HOME
[Environment]::SetEnvironmentVariable("JAVA_HOME", $javaPath, "User")
$env:JAVA_HOME = $javaPath

# Ajout au PATH utilisateur si ce n'est pas déjà fait
$currentPath = [Environment]::GetEnvironmentVariable("Path", "User")
if ($currentPath -notlike "*$javaPath\bin*") {
    [Environment]::SetEnvironmentVariable("Path", "$currentPath;$javaPath\bin", "User")
    $env:Path += ";$javaPath\bin"
}

Write-Host "✅ JAVA_HOME configuré avec succès !" -ForegroundColor Green
